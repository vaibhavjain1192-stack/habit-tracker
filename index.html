const SHEET_NAMES = {
  categories: 'Categories',
  habits:     'Habits',
  tracking:   'Tracking'
};

function makeResponse(data) {
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

function initSheets() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  function ensureSheet(name, headers) {
    let sheet = ss.getSheetByName(name);
    if (!sheet) {
      sheet = ss.insertSheet(name);
      sheet.getRange(1,1,1,headers.length).setValues([headers]);
      sheet.getRange(1,1,1,headers.length).setFontWeight('bold').setBackground('#1a1a2e').setFontColor('#ffffff');
      sheet.setFrozenRows(1);
    }
    return sheet;
  }
  ensureSheet(SHEET_NAMES.categories, ['id','name','color','created_at']);
  ensureSheet(SHEET_NAMES.habits,     ['id','category_id','category_name','name','created_at']);
  ensureSheet(SHEET_NAMES.tracking,   ['id','habit_id','habit_name','category_name','start_date','checked_days','created_at']);
}

function doGet(e) {
  try {
    initSheets();
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    function sheetToObjects(sheetName) {
      const sheet = ss.getSheetByName(sheetName);
      if (!sheet) return [];
      const data = sheet.getDataRange().getValues();
      if (data.length < 2) return [];
      const headers = data[0];
      return data.slice(1).map(row => {
        const obj = {};
        headers.forEach((h,i) => { obj[h] = row[i]; });
        return obj;
      }).filter(obj => obj.id);
    }
    const categories = sheetToObjects(SHEET_NAMES.categories).map(c => ({
      id:String(c.id), name:String(c.name||''), color:String(c.color||'#22c55e'), created_at:String(c.created_at||'')
    }));
    const habits = sheetToObjects(SHEET_NAMES.habits).map(h => ({
      id:String(h.id), category_id:String(h.category_id||''), category_name:String(h.category_name||''), name:String(h.name||''), created_at:String(h.created_at||'')
    }));
    const tracking = sheetToObjects(SHEET_NAMES.tracking).map(t => {
      let checked = [];
      try { checked = JSON.parse(String(t.checked_days||'[]')); } catch(e) {}
      return { id:String(t.id), habit_id:String(t.habit_id||''), habit_name:String(t.habit_name||''), category_name:String(t.category_name||''), start_date:String(t.start_date||''), checked_days:checked, created_at:String(t.created_at||'') };
    });
    return makeResponse({ ok:true, categories, habits, tracking });
  } catch(err) { return makeResponse({ ok:false, error:err.message }); }
}

function doPost(e) {
  try {
    initSheets();
    const payload = JSON.parse(e.postData.contents);
    const action  = payload.action;
    const data    = payload.data || {};
    const ss      = SpreadsheetApp.getActiveSpreadsheet();
    function genId() { return Date.now().toString(36) + Math.random().toString(36).slice(2,6); }
    function getSheet(name) { return ss.getSheetByName(name); }
    function appendRow(sn, row) { getSheet(sn).appendRow(row); }
    function findRowIndex(sn, id) {
      const sheet = getSheet(sn);
      const ids = sheet.getRange(1,1,sheet.getLastRow(),1).getValues().flat();
      return ids.indexOf(String(id));
    }
    function deleteRowById(sn, id) {
      const sheet = getSheet(sn);
      const idx = findRowIndex(sn, id);
      if (idx > 0) sheet.deleteRow(idx+1);
    }
    function updateCell(sn, id, col, val) {
      const sheet = getSheet(sn);
      const idx = findRowIndex(sn, id);
      if (idx > 0) sheet.getRange(idx+1, col).setValue(val);
    }
    const now = new Date().toISOString();
    if (action === 'add_category') {
      const id = genId(); appendRow(SHEET_NAMES.categories, [id, data.name, data.color, now]);
      return makeResponse({ ok:true, id });
    }
    if (action === 'delete_category') {
      const sheet = getSheet(SHEET_NAMES.habits);
      const values = sheet.getDataRange().getValues();
      for (let i = values.length-1; i >= 1; i--) {
        if (String(values[i][1]) === String(data.id)) sheet.deleteRow(i+1);
      }
      deleteRowById(SHEET_NAMES.categories, data.id);
      return makeResponse({ ok:true });
    }
    if (action === 'add_habit') {
      const id = genId(); appendRow(SHEET_NAMES.habits, [id, data.category_id, data.category_name, data.name, now]);
      return makeResponse({ ok:true, id });
    }
    if (action === 'delete_habit') { deleteRowById(SHEET_NAMES.habits, data.id); return makeResponse({ ok:true }); }
    if (action === 'start_tracking') {
      const id = genId();
      appendRow(SHEET_NAMES.tracking, [id, data.habit_id, data.habit_name, data.category_name, data.start_date, JSON.stringify([]), now]);
      return makeResponse({ ok:true, id });
    }
    if (action === 'toggle_day') { updateCell(SHEET_NAMES.tracking, data.id, 6, JSON.stringify(data.checked_days)); return makeResponse({ ok:true }); }
    if (action === 'update_start_date') {
      updateCell(SHEET_NAMES.tracking, data.id, 5, data.start_date);
      updateCell(SHEET_NAMES.tracking, data.id, 6, JSON.stringify(data.checked_days));
      return makeResponse({ ok:true });
    }
    if (action === 'remove_tracking') { deleteRowById(SHEET_NAMES.tracking, data.id); return makeResponse({ ok:true }); }
    return makeResponse({ ok:false, error:'Unknown action: '+action });
  } catch(err) { return makeResponse({ ok:false, error:err.message }); }
}
